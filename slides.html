SELECT grp, AVG(weekend_wait_min)
FROM donut_shops
WHERE grp IN ('A', 'B')
GROUP BY grp
HAVING COUNT(*) > 0;

FROM donut_shops d
JOIN walking_distances w
  ON w.shop_id = d.shop_id
WHERE w.origin = :me;
